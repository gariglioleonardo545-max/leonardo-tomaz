#include #include #include #include // ====================================================== // CONFIGURAÇÕES // ====================================================== // Wi-Fi do simulador Cirkit Designer const char* WIFI_SSID = "CirkitWifi"; const char* WIFI_PASSWORD = ""; // Seu Firebase Realtime Database const char* FIREBASE_URL = "https://leonardo-89179-default-rtdb.firebaseio.com/led.json"; // LED const int LED_PIN = 2; // Tempo entre consultas const unsigned long INTERVALO = 2000; unsigned long ultimaConsulta = 0; // ====================================================== // CONECTAR AO WIFI // ====================================================== void conectarWiFi() { Serial.println(); Serial.println("=============================="); Serial.println("CONECTANDO AO WIFI"); Serial.println("=============================="); WiFi.mode(WIFI_STA); Serial.print("Conectando ao CirkitWifi"); WiFi.begin(WIFI_SSID, WIFI_PASSWORD); int tentativas = 0; while (WiFi.status() != WL_CONNECTED) { delay(500); Serial.print("."); tentativas++; if (tentativas >= 20) { Serial.println(); Serial.println("ERRO: WIFI NAO CONECTOU!"); return; } } Serial.println(); Serial.println("WIFI CONECTADO!"); Serial.print("Endereco IP: "); Serial.println(WiFi.localIP()); Serial.println("=============================="); } // ====================================================== // LER FIREBASE // ====================================================== void lerFirebase() { if (WiFi.status() != WL_CONNECTED) { Serial.println("WiFi desconectado!"); conectarWiFi(); return; } Serial.println(); Serial.println("------------------------------"); Serial.println("CONSULTANDO FIREBASE..."); WiFiClientSecure client; // Para o simulador/teste client.setInsecure(); HTTPClient http; Serial.println("Abrindo conexao..."); if (!http.begin(client, FIREBASE_URL)) { Serial.println("ERRO: nao foi possivel abrir o Firebase!"); return; } Serial.println("Enviando GET..."); int codigoHTTP = http.GET(); Serial.print("Codigo HTTP: "); Serial.println(codigoHTTP); if (codigoHTTP == HTTP_CODE_OK) { String resposta = http.getString(); resposta.trim(); Serial.print("Firebase retornou: "); Serial.println(resposta); // ============================================== // TRUE = LIGA // ============================================== if (resposta == "true") { digitalWrite(LED_PIN, HIGH); Serial.println("COMANDO: LIGAR LED"); Serial.println("LED LIGADO!"); } // ============================================== // FALSE = DESLIGA // ============================================== else if (resposta == "false") { digitalWrite(LED_PIN, LOW); Serial.println("COMANDO: DESLIGAR LED"); Serial.println("LED DESLIGADO!"); } // ============================================== // VALOR INVALIDO // ============================================== else { Serial.println("ERRO: Firebase nao retornou true/false!"); } } else { Serial.println("ERRO AO ACESSAR FIREBASE!"); Serial.print("Descricao: "); Serial.println(http.errorToString(codigoHTTP)); } http.end(); Serial.println("------------------------------"); } // ====================================================== // SETUP // ====================================================== void setup() { Serial.begin(115200); delay(2000); Serial.println(); Serial.println(); Serial.println("################################"); Serial.println("# ESP32 + FIREBASE #"); Serial.println("################################"); // Configura LED pinMode(LED_PIN, OUTPUT); // LED começa desligado digitalWrite(LED_PIN, LOW); Serial.println("LED configurado no GPIO 2."); // Conecta ao WiFi conectarWiFi(); } // ====================================================== // LOOP // ====================================================== void loop() { if (millis() - ultimaConsulta >= INTERVALO) { ultimaConsulta = millis(); lerFirebase(); } }
